import java.util.*;

// Abstract base class
abstract class JobRole {
    private String candidateName;

    public JobRole(String candidateName) {
        this.candidateName = candidateName;
    }

    public String getCandidateName() {
        return candidateName;
    }

    public abstract void evaluate();
}

// Subclasses for each job role
class SoftwareEngineer extends JobRole {
    public SoftwareEngineer(String candidateName) {
        super(candidateName);
    }

    @Override
    public void evaluate() {
        System.out.println(getCandidateName() + " evaluated for Software Engineering skills.");
    }
}

class DataScientist extends JobRole {
    public DataScientist(String candidateName) {
        super(candidateName);
    }

    @Override
    public void evaluate() {
        System.out.println(getCandidateName() + " evaluated for Data Science skills.");
    }
}

class ProductManager extends JobRole {
    public ProductManager(String candidateName) {
        super(candidateName);
    }

    @Override
    public void evaluate() {
        System.out.println(getCandidateName() + " evaluated for Product Management skills.");
    }
}

// Generic Resume class
class Resume<T extends JobRole> {
    private T jobRole;

    public Resume(T jobRole) {
        this.jobRole = jobRole;
    }

    public void process() {
        System.out.println("Processing resume of: " + jobRole.getCandidateName());
        jobRole.evaluate();
    }

    public T getJobRole() {
        return jobRole;
    }
}

// Resume processor with wildcard method
class ResumeProcessor {
    public static void processAll(List<? extends JobRole> candidates) {
        for (JobRole jobRole : candidates) {
            System.out.println("Screening " + jobRole.getCandidateName());
            jobRole.evaluate();
        }
    }
}

// Main class to run the system
public class Main {
    public static void main(String[] args) {
        // Individual resume processing
        Resume<SoftwareEngineer> seResume = new Resume<>(new SoftwareEngineer("Alice"));
        Resume<DataScientist> dsResume = new Resume<>(new DataScientist("Bob"));
        Resume<ProductManager> pmResume = new Resume<>(new ProductManager("Charlie"));

        seResume.process();
        dsResume.process();
        pmResume.process();

        // Wildcard-based batch screening
        List<JobRole> allCandidates = Arrays.asList(
            new SoftwareEngineer("David"),
            new DataScientist("Eva"),
            new ProductManager("Frank")
        );

        System.out.println("\n--- Batch Resume Screening ---");
        ResumeProcessor.processAll(allCandidates);
